# 指令系统接口设计

## 概述

本文档定义了MVP版本游戏运行时引擎的指令系统接口。所有接口采用TypeScript定义，确保类型安全和开发体验。

## 核心接口定义

### 1. 基础指令接口

```typescript
/**
 * 游戏指令基础接口
 * 所有具体指令都必须实现此接口
 */
interface IGameCommand {
  /** 指令唯一标识符 */
  readonly id: string;
  
  /** 指令类型 */
  readonly type: CommandType;
  
  /** 指令参数 */
  readonly params: Record<string, any>;
  
  /** 执行条件（可选） */
  readonly conditions?: ConditionConfig[];
  
  /** 延迟执行时间（毫秒） */
  readonly delay?: number;
  
  /** 是否并行执行 */
  readonly parallel?: boolean;
  
  /** 指令优先级 */
  readonly priority?: number;
  
  /** 指令描述（用于调试） */
  readonly description?: string;
}

/**
 * 指令类型枚举
 */
type CommandType = 
  // 状态控制
  | 'set_variable' | 'add_variable' | 'set_switch' | 'toggle_switch'
  // 流程控制
  | 'wait' | 'jump_to' | 'call_event' | 'return' | 'stop'
  // 条件分支
  | 'if_condition' | 'else' | 'end_if' | 'loop' | 'break_loop'
  // 显示控制
  | 'show_image' | 'hide_image' | 'show_text' | 'hide_text'
  // 移动动画
  | 'move_to' | 'move_by' | 'rotate_to' | 'scale_to'
  // 音频控制
  | 'play_sound' | 'stop_sound' | 'play_music' | 'stop_music'
  // 用户交互
  | 'enable_click' | 'disable_click' | 'show_choices' | 'input_text'
  // 元素交互
  | 'set_draggable' | 'set_clickable' | 'set_droppable' | 'check_drop_zone'
  // 位置控制
  | 'set_position' | 'get_position' | 'check_collision' | 'check_in_area'
  // 特效动画
  | 'fade_in' | 'fade_out' | 'shake' | 'flash' | 'tint'
  // 游戏逻辑
  | 'check_answer' | 'add_score' | 'next_level' | 'game_over'
  // 自定义扩展
  | string;
```

### 2. 指令执行器接口

```typescript
/**
 * 指令执行器接口
 */
interface ICommandExecutor {
  /**
   * 注册指令处理器
   * @param type 指令类型
   * @param handler 指令处理器
   */
  registerCommand(type: CommandType, handler: CommandHandler): void;
  
  /**
   * 注销指令处理器
   * @param type 指令类型
   */
  unregisterCommand(type: CommandType): void;
  
  /**
   * 执行单个指令
   * @param command 要执行的指令
   * @returns 执行结果
   */
  executeCommand(command: IGameCommand): Promise<CommandResult>;
  
  /**
   * 执行指令序列
   * @param commands 指令数组
   * @returns 执行结果数组
   */
  executeCommandSequence(commands: IGameCommand[]): Promise<CommandResult[]>;
  
  /**
   * 并行执行指令
   * @param commands 指令数组
   * @returns 执行结果数组
   */
  executeCommandParallel(commands: IGameCommand[]): Promise<CommandResult[]>;
  
  /**
   * 验证指令
   * @param command 要验证的指令
   * @returns 验证结果
   */
  validateCommand(command: IGameCommand): ValidationResult;
  
  /**
   * 获取已注册的指令类型
   * @returns 指令类型数组
   */
  getRegisteredCommands(): CommandType[];
  
  /**
   * 检查指令是否已注册
   * @param type 指令类型
   * @returns 是否已注册
   */
  hasCommand(type: CommandType): boolean;
}

/**
 * 指令处理器函数类型
 */
type CommandHandler = (
  command: IGameCommand,
  context: ICommandContext
) => Promise<CommandResult> | CommandResult;
```

### 5. 条件配置接口

```typescript
/**
 * 条件配置接口
 */
interface ConditionConfig {
  /** 条件类型 */
  readonly type: ConditionType;
  
  /** 比较操作符 */
  readonly operator: ComparisonOperator;
  
  /** 左操作数 */
  readonly left: ConditionValue;
  
  /** 右操作数 */
  readonly right: ConditionValue;
  
  /** 逻辑连接符（用于组合条件） */
  readonly logic?: LogicOperator;
  
  /** 下一个条件（用于链式条件） */
  readonly next?: ConditionConfig;
  
  /** 条件描述 */
  readonly description?: string;
}

/**
 * 条件类型
 */
type ConditionType = 
  | 'variable'    // 变量比较
  | 'switch'      // 开关状态
  | 'score'       // 分数比较
  | 'time'        // 时间比较
  | 'level'       // 关卡比较
  | 'resource'    // 资源状态
  | 'position'    // 位置条件
  | 'distance'    // 距离条件
  | 'collision'   // 碰撞检测
  | 'area'        // 区域检测
  | 'element'     // 元素状态
  | 'custom'      // 自定义条件
  | 'expression'; // 表达式求值

/**
 * 比较操作符
 */
type ComparisonOperator = 
  | '=='          // 等于
  | '!='          // 不等于
  | '>'           // 大于
  | '<'           // 小于
  | '>='          // 大于等于
  | '<='          // 小于等于
  | 'contains'    // 包含
  | 'startsWith'  // 以...开始
  | 'endsWith'    // 以...结束
  | 'matches'     // 正则匹配
  | 'exists'      // 存在
  | 'empty';      // 为空

/**
 * 逻辑操作符
 */
type LogicOperator = 'and' | 'or' | 'not';

/**
 * 条件值类型
 */
type ConditionValue = 
  | string        // 字符串字面量
  | number        // 数字字面量
  | boolean       // 布尔字面量
  | VariableRef   // 变量引用
  | SwitchRef     // 开关引用
  | PositionRef   // 位置引用
  | ElementRef    // 元素引用
  | Expression;   // 表达式

/**
 * 变量引用
 */
interface VariableRef {
  readonly type: 'variable';
  readonly name: string;
}

/**
 * 开关引用
 */
interface SwitchRef {
  readonly type: 'switch';
  readonly name: string;
}

/**
 * 表达式
 */
interface Expression {
  readonly type: 'expression';
  readonly code: string;
}

/**
 * 位置引用
 */

  readonly elementId: string;

  readonly property: 'visible' | 'interactive' | 'draggable' | 'clickable' | 'state';
}
```

### 12. 指令装饰器

```typescript
/**
 * 指令装饰器接口
 */
interface ICommandDecorator {
  /**
   * 装饰指令执行前
   * @param command 原始指令
   * @param context 执行上下文
   */
  beforeExecute(command: IGameCommand, context: ICommandContext): Promise<void> | void;
  
  /**
   * 装饰指令执行后
   * @param command 原始指令
   * @param result 执行结果
   * @param context 执行上下文
   */
  afterExecute(
    command: IGameCommand, 
    result: CommandResult, 
    context: ICommandContext
  ): Promise<CommandResult> | CommandResult;
  
  /**
   * 处理执行错误
   * @param command 原始指令
   * @param error 错误对象
   * @param context 执行上下文
   */
  onError(
    command: IGameCommand, 
    error: Error, 
    context: ICommandContext
  ): Promise<CommandResult> | CommandResult;
}

/**
 * 常用装饰器
 */
class LoggingDecorator implements ICommandDecorator {
  beforeExecute(command: IGameCommand, context: ICommandContext): void {
    context.log('debug', `Executing command: ${command.type}`, { command });
  }
  
  afterExecute(
    command: IGameCommand, 
    result: CommandResult, 
    context: ICommandContext
  ): CommandResult {
    context.log('debug', `Command completed: ${command.type}`, { result });
    return result;
  }
  
  onError(
    command: IGameCommand, 
    error: Error, 
    context: ICommandContext
  ): CommandResult {
    context.log('error', `Command failed: ${command.type}`, { command, error });
    return {
      success: false,
      command,
      startTime: Date.now(),
      endTime: Date.now(),
      duration: 0,
      error,
      stateChanges: [],
      triggeredEvents: [],
      logs: []
    };
  }
}

class PerformanceDecorator implements ICommandDecorator {
  private startTime: number = 0;
  
  beforeExecute(command: IGameCommand, context: ICommandContext): void {
    this.startTime = performance.now();
  }
  
  afterExecute(
    command: IGameCommand, 
    result: CommandResult, 
    context: ICommandContext
  ): CommandResult {
    const duration = performance.now() - this.startTime;
    if (duration > 100) { // 超过100ms记录警告
      context.log('warn', `Slow command execution: ${command.type}`, { duration });
    }
    return result;
  }
  
  onError(): CommandResult {
    // 错误情况下返回默认结果
    return {
      success: false,
      command: {} as IGameCommand,
      startTime: this.startTime,
      endTime: performance.now(),
      duration: performance.now() - this.startTime,
      stateChanges: [],
      triggeredEvents: [],
      logs: []
    };
  }
}
```

## 使用示例

### 14. 基础使用示例

```typescript
// 创建指令执行器
const executor = new CommandExecutor(stateManager, eventManager);

// 注册自定义指令
executor.registerCommand('custom_action', async (command, context) => {
  // 自定义指令逻辑
  context.log('info', 'Executing custom action');
  return {
    success: true,
    command,
    startTime: Date.now(),
    endTime: Date.now(),
    duration: 0,
    stateChanges: [],
    triggeredEvents: [],
    logs: []
  };
});

// 执行指令
const command: SetVariableCommand = {
  id: 'cmd_1',
  type: 'set_variable',
  params: {
    variableName: 'score',
    value: 100,
    operation: 'set'
  }
};
```
